name: AI_BOM Check
on:
  push:
    branches: [ main ]  # Adjust this to your master branch name
  pull_request:
    branches: [ main ]  # Optionally run on PRs
jobs:
  ai_bom-check:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262 # v4.4.0

      # Everything but .git is scanned (vendored code, build output and nested archives included); installed
      # dependency trees stay out. The zip is written outside the workspace. Uploads are capped at 50 MB.
      - name: Create project zip
        run: |
          zip -q -r "$RUNNER_TEMP/project.zip" . -x '.git/*' '*/.git/*' 'node_modules/*' '*/node_modules/*' 'venv/*' '.venv/*' '*/.venv/*'

      - name: Perform AI_BOM Scan
        id: ai_bom
        env:
          CICD_API_KEY: ${{ secrets.CICD_API_KEY }}
        run: |
          set -u
          # A failed scan fails the pipeline: --fail-with-body (curl 7.76+) exits non-zero on any HTTP error and still
          # prints the reason. 429 (rate limited): wait Retry-After seconds (30 if absent, 300 at most), up to 5 tries.
          for attempt in 1 2 3 4 5; do
            rc=0
            code=$(curl --fail-with-body -sS --max-time 900 -o response.json -D devsecshield-headers.txt -w '%{http_code}' \
              -H "cicd-api-key: $CICD_API_KEY" -F "zip_file=@$RUNNER_TEMP/project.zip" \
              "https://api.example.test/cicd/ai_bom/scan") || rc=$?
            if [ "${code}" = "429" ] && [ "${attempt}" -lt 5 ]; then
              delay=$(awk 'tolower($1) == "retry-after:" {print $2 + 0}' devsecshield-headers.txt | tail -n 1)
              [ -n "${delay}" ] && [ "${delay}" -gt 0 ] || delay=30
              [ "${delay}" -le 300 ] || delay=300
              echo "DevSecShield is rate limiting this key; retrying in ${delay}s (attempt ${attempt} of 5)"
              sleep "${delay}"
              continue
            fi
            break
          done
          if [ "${rc}" -ne 0 ]; then
            echo "DevSecShield scan failed (curl exit ${rc}, HTTP ${code:-none}):"
            cat response.json 2>/dev/null || true
            exit 1
          fi
          echo "AI_BOM findings: $(jq '.finding_count // 0' response.json)"
          jq -r '(.results | if type == "object" then (.results // []) elif type == "array" then . else [] end)[] | (.impact // "unknown") + " " + (.title // .name // "")' response.json
